member-app-logging
    - all-logs-member-app,q:{service_name="member-app"} |= ""
    - log-volume-member-app,q:count_over_time({service_name="member-app"}[5m])
    - error-rate-member-app,q:count_over_time({service_name="member-app"} |= "error" [5m])
    - top-error-message-member-app,q:topk(10, count_over_time({service_name="member-app"} |= "error" [5m]))

member-app-metrics
    - cpu-usage-member-app,q:avg(system_cpu_usage_percent{job="member-app"})
    - memory-usage-member-app,q:(system_memory_usage_bytes{job="member-app"}/system_memory_total_bytes{job="member-app"})*100
    - disk-usage-member-app,q:(system_disk_usage_bytes{job="member-app"}/system_disk_total_bytes{job="member-app"})*100
